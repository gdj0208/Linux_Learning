# 시스템 V의 프로세스 간 통신

## 1️⃣ 시스템 V IPC

### 🎯 시스템 V IPC
- 유닉스는 BSD계열과 시트템V 계열로 구분됨
- 시스템 V IPC : 메시지큐 + 공유 메모리 + 세마포어
- 시스템 V 계열 유닉스에서 개발해 제공하는 프로세스 간 통신법


### 🎯 IPC 객체
- 시스템 V IPC를 사용하려면 IPC 객체를 생성해야함
- IPC 객체 생성을 위해 **키**와 **식별자** 사용

**[ IPC 객체 관리 명령 ]**
```c
IPC_PRIVATE
key_t ftok(const char *pathname, int proj_id);
```

```bash
ipcs            # IPC 정보 검색
ipcmk [options] # IPC 생성
ipcrm [options] # IPC 삭제
```



### 🎯 메시지 큐 
**[ 특서 ]**
- 파이프와 유사
- 메시지(패킷) 단위로 동작
- 우편함처럼 메시지 큐를 만든 다음 이로 메시지를 주고 받음

**[ 예시 코드 ]**
```c
int msgget(key_t key, int msgflg);                                  // 메시지 큐 식별자 생성
int msgsnd(int msqid, const void *msgp, size_t msgsz, int msgflg);  // 메시지 전송
ssize_t msgrcv(int msqid, void *msgp, size_t msgsz, long msgtyp, int msgflg) // 메시지 수신
int msgctl(int msqid, int cmd, struct msqid_ds *buf)                // 메시지 제어
```



### 🎯 공유 메모리
**[ 공유 메모리 함수 ]**
- 한 프로세스의 일부분들 다른 프로세스와 공유
- 메모리의 일부 공간을 두 독립적인 프로세스 공유 -> 해당 메모리를 통해 데이터를 주고 받음

**[ 공유 메모리 함수 ]**
```c
int shmget(key_t key, size_t size, int shmflg);             // 공유 메모리 식별자 생성
void *shmat(int shmid, const void *shmaddr, int shmflg);    // 공유 메모리 연결
int shmdt(const void *shmaddr);                             // 공유 메모리 연결 해제
int shmctl(int shmid, int cmd, struct shmid_ds *bug);       // 공유 메모리 제어
```


### 🎯 세마퐅어
**[ 세마포어 정의 ]**
- 프로세스 사이의 동기를 맞추는 기능
- 공유 메모리에 여러 프로세스가 동시에 쓰기를 시도하면 데이터 손상
- 세마 포어가 복수의 프로세스 사이에 동작 순서를 제공

**[ 세마포어 함수 ]**
```c
int semget(key_t key, int nsems, int semflg);               // 세마포어 생성
int semctl(int semid, int semnum, int cmd, ...);            // 세마포어 제어
int semop(int semid, struct sembuf *sops, size_t nsops);    // 세마포어 연산
```





## 2️⃣시스템 V IPC의 공통 요소

### 🎯 키와 식별자
1. 키로 `IPC_PIVATE` 지정
    - 생성된 식별자를 서버와 클라이언트 모두 알 수 있게 생성
    - `fork()`로 생성된 부모-자식 프로세스 간 통신에서 유용하게 사용

2. `ftok()`로 키 생성
    - 경로명과 숫자값을 받아서 key를 생성
    - 서버와 클라이언트가 같은 경로명과 숫자값을 지정하면 공통 식별자 생성 가능

3. key 생성의 주의 사항
    - 같은 키로 생성된 식별자는 통신에 사용할 수 있음
    - 미리 정해진 key를 서버와 클라이언트 프로세스가 공유



### 🎯 키 생성하기 : ftok()
**[ 코드 ]**
```c
#include <sys/types.h>
#include <sys/ipc.h>

key_t ftok(const char *pathname, int proj_id);
```
- `pathname` : 파일 시스템에 지정한 경로명과 `proj_id`에 지정된 정숫값을 조합해 `key`를 생성 
    - 경로명 + 접근 권한 필요
- `proj_id` : `key`값을 생성할 때 지정하는 임의의 번호

**[ IPC 공통 구조체 ]**
```c
struct ipc_perm {
    key_t key;              // key 값
    uid_t uid;              // 구조체 소유자 ID
    gid_t gid;              // 구조체 소유 그룹 ID
    uid_t cuid;             // 구조체를 생성한 소유자 ID    
    gid_t cgid;             // 구조체를 생성한 그룹 ID
    unsigned short mode;    // 구조체에 대한 접근 권한
    unsigned short _seq;    // 일련 번호
}
```


### 🎯 시스템 V IPC 정보 검색
**[ 명령의 기본 형식 ]**
- ipcs : 시스템V IPC의 정보를 검색하고 현재 상태를 확인하는 명령
- ipcs 명령을 실행하는 동안에도 IPC 상태가 변경될 수 있음
- ipcs 명령은 검색하는 순간의 정확성만 보장

**[ 옵션 ]**
```bash
# 일반 옵션
-i id   # 특정 요소에 대한 정보를 출력
-h      # 도움말 출력
-V      # 버전

# 자원 옵션
-m      # 공유 메모리 정보 검색
-q      # 메시지 큐 정보 검색
-s      # 세마포어 정보 검색
-a      # 공유 메모리, 메시지큐, 세마포어 모두의 정보를 검색
```


### 🎯 시스템 V IPC 자원의 생성과 삭제
**[ ipcmk 명령 ]**
```bash
ipcmk [options]
```
- `-M size` : size에 지정한 byte 크기로 메모리 생성
- `-Q` : 메시지 큐 작성
- `-S number` : number에 해당되는 세마포어 생성
- `-p mode` : 자원 접근 권한 (0644)

**[ ipcrm 명령 ]**
```c
ipcrm [options]
```
- `-a` : 모든 자원 제거
- `-M shmkey` : shmeky로 생성한 공유 메모리의 **마지막 연결이 해제**된 후 **공유메모리 제거**
- `-m shmid` : shmid로 지정한 공유 메모리 삭제
- `-Q msgkey` : msgkey로 생성한 메시지 큐 제거
- `-q msgid` : msgid로 지정한 메시지 큐 삭제
- `-S semkey` : semkey로 생성한 세마포어 삭제
- `-s semid` : semid로 지정한 세마포어 삭제





## 3️⃣ 메시지 큐

### 🎯 메시지 큐 식별자 생성 : msgget()
**[ 코드 ]**
```c
#include <sys/types.h>
#include <sys/ipc.h>
#include <sys/msg.h>

int msgget(key_t key, int msgflg);
```
- `key` : 메시지 큐를 구별하는 key
    - `IPC_PRIVATE` 또는 `ftok()`
- `msgflg` : 메시지 큐의 속성을 설정하는 플래그
    - `IPC_CREAT` : 식별자를 새로 생성
    - `IPC_EXCEL` : 이미 존재하는 키면 오류 발생

**[ msqid_ds 구조체 ]**
```c
struct msqid_ds {
    struct ipc_perm msg_perm;   // IPC 공통 구조체
    time_t msg_stime;           // 마지막으로 메시지를 보낸 시각
    time_t msg_rtime;           // 마지막으로 메시지를 읽은 시각
    time_t msg_ctime;           // 마지막으로 메시지 큐의 권한을 바꾼 시각
    unsigned long __msg_cbytes; // 현 메시지 큐에 있는 총 바이트 수
    msqnum_t msq_qnum;          // 메시지 큐에 있는 메시지 개수
    msglen_t msg_qbytes;        // 메시지 큐의 최대 크기
    pid_t msg_lspid;            // 마지막으로 메시지를 보낸 프로세스의 PID
    pid_t msg_lrpid;            // 마지막으로 메시지를 읽은 프로세스의 PID
}
```

**[ 코드 예시 ]**
```c
key_t key;
int id;

key = ftok("keyfile", 1);
id = msgget(key, IPC_CREAT | 0640);
```


### 🎯 메시지 전송 : mssgsnd()
**[ 코드 ]**
```c
#include <sys/types.h>
#include <sys/ipc.h>
#include <sys/msg.h>

int msgsnd(int msqid, const void *msgp, size_t msgsz, int msgflg);
```
- `msqid` : msgget 함수로 생성한 메시지큐 식별자
- `msgp` : 메시지를 담고 있는 메시지 버퍼의 주소
- `msgsz` : 메시지의 크기
- `msgflg` : 블록 모드(0) / 비블록 모드(IPC_NOWAIT)

**[ msgbuf 구조체 ]**
```c
struct msgbuf {
    long mtype;     // 메시지 유형으로 양수를 지정
    char mtext[1];  // msgsz로 지정한 크기의 버퍼로 메시지 내용이 저장
}
```

**[ 메시지 큐를 생성하고 전송하는 예시 ]**
```c
// e1_client.c
#include <fcntl.h>
#include <unistd.h>
#include <stdlib.h>
#include <stdio.h>

int main() {
    int pd, n;
    char inmsg[80];

    if ((pd = open("./TEST-FIFO", O_RDONLY)) == -1) {
        perror("open");
        exit(1);
    }

    printf("Client =====\n");
    write(1, "From Server :", 13);
    while ((n=read(pd, inmsg, 80)) > 0)
        write(1, inmsg, n);

    if (n == -1) {
        perror("read");
        exit(1);
    }

    write(1, "\n", 1);
    close(pd);
}
```

```c
// e1_server.c
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>
#include <stdlib.h>
#include <stdio.h>
#include <string.h>

int main() {
    int pd, n;
    char msg[] = "Hello, FIFO";

    printf("Server =====\n");

    if (mkfifo("./TEST-FIFO", 0666) == -1) {
        perror("mkfifo");
        exit(1);
    }
    if ((pd = open("./TEST-FIFO", O_WRONLY)) == -1) {
        perror("open");
        exit(1);
    }

    printf("To Client : %s\n", msg);

    n = write(pd, msg, strlen(msg)+1);
    if (n == -1) {
        perror("write");
        exit(1);
    }
    close(pd);
}
```

### 🎯 메시지 수신 : msgrcv()
**[ 코드 ]**
```c
#include <sys/types.h>
#include <sys/ipc.h>
#include <sys/msg.h>

ssize_t msgrcv(
    int msqid,      // 메시지 큐 식별자
    void *msgp,     // 메시지 버퍼 주소
    size_t msgsz,   // 메시지 버퍼 크기
    long msgtyp,    // 읽어올 메시지 유형
    int msgflg      // 블록 모드(0), 비볼록 모드(IPC_NOWAIT)
);
```
- `msgtyp`에 지정 가능한 값 : 읽어올 메시지 유형
    - 0 : 메시지 큐의 가장 앞에 있는 메시지
    - 양수 : 메시지 큐에서 msgtyp로 지정한 유형과 같은 메시지를 읽어옴
    - 음수 : 메시지 큐에서 msgtyp로 지정한 값의 절대값이 같거나 작은 메시지를 읽어옴

**[ 메시지 수신 예시 ]**
```c
// e2_client.c
#include <sys/msg.h>
#include <stdlib.h>
#include <stdio.h>
#include <string.h>

struct mymsgbuf {
    long mtype;
    char mtext[80];
};

int main() {
    key_t key;
    int msgid;
    struct mymsgbuf mesg;

    //key = ftok("keyfile", 1);
    msgid = msgget(0777, IPC_CREAT|0644);
    if (msgid == -1) {
        perror("msgget");
        exit(1);
    }

    mesg.mtype = 1;
    strcpy(mesg.mtext, "Message Q Test");

    if (msgsnd(msgid, (void *)&mesg, 80, IPC_NOWAIT) == -1) {
        perror("msgsnd");
        exit(1);
    }
}
```

```c
// e2_server.c
#include <sys/msg.h>
#include <stdlib.h>
#include <stdio.h>

struct mymsgbuf {
    long mtype;
    char mtext[80];
};

int main() {
    struct mymsgbuf inmsg;
    key_t key;
    int msgid, len;

    //key = ftok("keyfile", 1);
    if ((msgid = msgget(0777, 0)) < 0) {
        perror("msgget");
        exit(1);
    }

    len = msgrcv(msgid, &inmsg, 80, 0, 0);
    printf("Received Msg = %s, Len=%d\n", inmsg.mtext, len);
}
```


### 🎯 메시지 제어 : msgctl
**[ 코드 ]**
```c
#include <sys/types.h>
#include <sys/ipc.h>
#include <sys/msg.h>

int msgctl(
    int msqid,              // 메시지 큐 식별자
    int cmd,                // 수행할 제어 기능
    struct msqid_ds *buf    // 메시지 큐 구조체의 주서
);
```
- `cmd` 지정 가능 목록
    - `IPC_STAT` : 현재 메시지 큐의 정보를 buf로 지정한 메모리에 저장
    - `IPC_SET` : 메시지 큐 정보값을 변경
    - `IPC_RMID` : msqid로 지정한 메시지 큐 제거, 관련된 데이터 구조체 제거
    - `IPC_INFO` : 메시지 큐의 제한값을 vuf에 저장

**[ msginfo 구조체 ]**
```c
struct msginfo {
    int msgpool;            // 메시지 데이터를 저장할 버퍼 공간 크기
    int msgmap;             // 메시지 맵의 최대 항목 수
    int msgmax;             // 메시지에 저장할 수 있는 메시지의 최대 크기
    int msgmnb;             // 메시지 큐에 작성할 수 있는 최대 크기
    int msgmni;             // 메시지 큐의 최대 개수
    int msgssz;             // 메시지 세그먼트 크기
    int msgtql;             // 시스템에 잇는 모든 메시지 큐의 최대 메시지 수
    unsigned short msgseq;  // 세그먼트 최대 개수
}
```

## 4️⃣ 공유 메모리

### 🎯 공유 메모리 식별자 생성 : shmget()
**[ 설명 ]**
```c
#include <sys/ipc.h>
#include <sys/shm.h>

int shmget(key_t key, size_t size, int shmflg);
```
- key : IPC_PRIVATE 또는 ftok()로 생성한 키
- size : 공유할 메모리 크기
- shmflg : 공유 메모리의 속성을 지정하는 플래그
    - IPC_CREAT
    - IPC_EXCL

**[ shmid_ds 객체 ]**
```c
struct shmid_ds {
    struct ipc_perm msg_perm;   // IPC 공통 구조체
    size_t shm_segsz;           // 공유 메모리 세그먼트의 크기
    time_t shm_atime;           // 마지막으로 공유 메모리에 연결한 시각
    time_t shm_dtime;           // 마지막으로 공유 메모리의 연결을 해제한 시각
    time_t shm_ctime;           // 마지막으로 공유 메모리의 접근 권한을 변경한 시각
    pid_t shm_lpid;             // 마지막으로 shmop 동작을 한 프로세스의 PID
    pid_t shm_cpid;             //공유 메모리를 생성한 프로세스의 PID
    shmat_t shm_nattch;         // 공유 메모리를 연결하고 있는 프로세스의 수
}
```

### 🎯 공유 메모리 연결 : shmat()
**[ 코드 ]**
```c
#include <sys/types.h>
#include <sys/shm.h>

void *shmat(int shmid, const void *shmaddr, int shmflg);
```
- shmid : shmget()로 생성한 공유 메모리 식별자
    - 주로 NULL 입력
    - 시스템이 알아서 적절한 주소에 공유 메모리를 연결
- shmaddr : 공유 메모리를 연결할 주소
- shmflg : 공유 메모리에 대한 읽기/쓰기 권한



### 🎯 공유 메모리 연결 해제 : shmdt()
**[ 코드 ]**
```c
#include <sys/types.h>
#include <sys/shm.h>

void *shmdt(const void *shmaddr);
```
- shmaddr : 연결을 해제할 송유 메모리의 시작 주소



### 🎯 공유 메모리 제정 : shmctl()
**[ 코드 ]**
```c
#include <sys/ipc.h>
#include <sys/shm.h>

int shmctl(int shmid, int cmd, struct shmid_ds *buf);
```
- shmid : shmget() 함수로 생성한 공유 메모리 식별자
- cmd : 수행할 제어 기능
- buf : 제어 기능에 사용되는 공유 메모리 구조체 주소
    - IPC_STAT
    - IPC_SET
    - IPC_RMID

## 5️⃣ 세마포어

### 🎯 세마포어 기본 개념
- 중요 처리 부분에 잠금 기능 설정
    ```
    p(sem);             // 잠금
    /* 중요 처리 부분 */
    v(sem);             // 잠금 해제
    ```

1. p()의 기본 동작 구조
    ```
    p(sem) {
        while sem==0 do wait;
        sem -= 1
    }
    ```
    - sem이 0이면 다른 프로세스가 처리 부분을 수행하고 있다는 것
    - sem이 1이 될 때까지 대기

2. v()의 기본 동작 구조
    ```
    v(sem) {
        sem += 1
        if(대기중인 프로세스 존재)
            대기중인 첫 번째 프로세스를 동작
    }
    ```



### 🎯 세마포어 구조체
```
struct semid_ds {
    struct ipc_perm sem_per;    // 소유자 접근 권한 정보
    ushort_t sem_nsems;         // 집합 속 세마포어의 갯수
    time_t sem_ctime;
    time_t sem_otime;           
}
```
- 세마포어는 단순히 정수 하나가 아닌 세마포어들의 집합으로 관리된다.


### 🎯 세마포어 연관 값
- semval : 세마포어 값
- sempid : 세마포어에 최근 접근했던 프로세스 pid
- semncnt : 세마포어의 값이 현재보다 더 큰 값을 갖기를 기다리는 프로세스 수
- semzcnt : 세마포어의 값이 0이 되길 기다리는 프로세스 수



### 🎯 세마포어 생성 : semget()
```
#include <sys/types.h>
#include <sys/ipc.h>
#include <sys/sem.h>

int semget(key_t key, int nsems, int semflg);
```
- key : IPC_PRIVATE 또는 ftok()로 생성한 키
- nsems : 생성할 세마포어 수
- semflg : 세마포어 접근 속성 (IPC_CREAT, IPC_EXCL)



### 🎯 세마포어 제어 : semctl()
```
int semctl(int semid, int semnum, int cmd, union semun);
```
- semid : semget()로 생성한 세마포어 식별자
- semnum : 기능을 제어할 세마포어 번호
- cmd : 수행할 제어 명령 ( IPC_STAT, IPC_SET, IPC_RMID )
- semun : 제어 명령에 따라 필요한 세마포어 공용체 주소
    ```
    union semun {
        int val;
        struct semid_ds * buf;
        unsigned short *array;
        struct seminfo *__buf;
    }
    ```



### 🎯 세마포어 연산 : semop()
- 세마포어 값을 변경하여 잠그거나(Wait/P) 푼다(Signal/V).
```
#include <sys/types.h>
#include <sys/ipc.h>
#include <sys/sem.h>

int semop(int semid, struct sembuf *sops, size_t nsops);
```
- sedmid : semget()로 생성한 세마포어 식별자 아이디 
- sops : sembuf 구조체 주소
    ```
    struct sembuf {
        unsigned short sem_num; // 세마포어 번호
        short sem_op;           // 세마포어 연산
        short sem_flg;          // IPC_NOWAIT, SEM_UNDO(프로세스 비정상 종료시 세마포어 동작 취소)
    }
    ```
- nsops : sops의 크기

- semop() : 아래의 코드처럼 동작함 
    ```
    if (sem_op < 0) {                           // 세마포어 잠금
        wait until semval >= |semp_op|
        semval -= |sem_op|
    }
    else if(sem_op > 0) { semval += sem_op; }   // 세마포어 잠금 해제
    else { wait untill semval == 0 }
    ```

    

### 🎯 세마포어 예시
```
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/ipc.h>
#include <sys/sem.h>
#include <sys/wait.h>
#include <errno.h>

// semctl()을 위한 공용체 정의 (리눅스 맨페이지 권장사항)
union semun {
    int val;
    struct semid_ds *buf;
    unsigned short *array;
};

// P 연산 (잠금 / Wait): 세마포어 값을 1 뺌
void p(int semid) {
    struct sembuf pbuf;
    pbuf.sem_num = 0;        // 0번째 세마포어 대상
    pbuf.sem_op = -1;        // 값을 1 감소 (0이면 대기)
    pbuf.sem_flg = SEM_UNDO; // 프로세스 종료 시 자동 해제
    
    if (semop(semid, &pbuf, 1) == -1) {
        perror("p operation failed");
        exit(1);
    }
}

// V 연산 (해제 / Signal): 세마포어 값을 1 더함
void v(int semid) {
    struct sembuf vbuf;
    vbuf.sem_num = 0;
    vbuf.sem_op = 1;         // 값을 1 증가 (대기자 깨움)
    vbuf.sem_flg = SEM_UNDO;

    if (semop(semid, &vbuf, 1) == -1) {
        perror("v operation failed");
        exit(1);
    }
}

int main() {
    int semid;
    pid_t pid;
    union semun arg;
    
    // 1. 세마포어 집합 생성 (키: IPC_PRIVATE, 개수: 1개)
    if ((semid = semget(IPC_PRIVATE, 1, IPC_CREAT | 0666)) == -1) {
        perror("semget failed");
        exit(1);
    }

    // 2. 세마포어 값 초기화 (SETVAL)
    // 값을 1로 설정하여 "Binary Semaphore(Mutex)" 처럼 사용
    arg.val = 1; 
    if (semctl(semid, 0, SETVAL, arg) == -1) {
        perror("semctl init failed");
        exit(1);
    }

    // 3. 프로세스 분기
    pid = fork();

    if (pid < 0) {
        perror("fork failed");
        exit(1);
    }
    else if (pid == 0) { 
        // [자식 프로세스]
        printf("    [자식] 임계 구역 진입 시도...\n");
        p(semid); // 잠금 (Count: 1 -> 0)
        
        printf("    [자식] 임계 구역 진입! (작업 중...)\n");
        sleep(2); // 작업 시뮬레이션
        printf("    [자식] 임계 구역 퇴장.\n");
        
        v(semid); // 해제 (Count: 0 -> 1)
        exit(0);
    }
    else {
        // [부모 프로세스]
        printf("[부모] 임계 구역 진입 시도...\n");
        p(semid); // 잠금
        
        printf("[부모] 임계 구역 진입! (작업 중...)\n");
        sleep(2); // 작업 시뮬레이션
        printf("[부모] 임계 구역 퇴장.\n");
        
        v(semid); // 해제
        
        wait(NULL); // 자식 종료 대기

        // 4. 세마포어 삭제 (IPC_RMID)
        // 다 쓴 세마포어는 반드시 삭제해야 함!
        if (semctl(semid, 0, IPC_RMID) == -1) {
            perror("semctl remove failed");
            exit(1);
        }
        printf("세마포어 삭제 완료.\n");
    }

    return 0;
}
```
